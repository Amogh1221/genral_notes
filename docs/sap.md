### 1.1 Count elements not common to both lists
**Problem:** Given two lists L1, L2, find how many elements appear in exactly one of the two (symmetric difference count).
Example: L1=[1,2,3,4,8,9,10], L2=[11,12,13,14,4,5,8,9,8] → Output: 9

```cpp
int countNotCommon(vector<int>& L1, vector<int>& L2) {
    unordered_set<int> s1(L1.begin(), L1.end());
    unordered_set<int> s2(L2.begin(), L2.end());
    int count = 0;
    for (int x : s1) if (!s2.count(x)) count++;
    for (int x : s2) if (!s1.count(x)) count++;
    return count;
}
```
**Complexity:** O(n + m).

### 1.2 Maximum points on the same line
**Problem:** Given X[] and Y[] representing points, find the max number of points lying on one straight line.

```cpp
int gcdSafe(int a, int b) {
    a = abs(a); b = abs(b);
    while (b) { a %= b; swap(a, b); }
    return a;
}

int maxPoints(vector<int>& X, vector<int>& Y) {
    int n = X.size();
    if (n <= 2) return n;
    int best = 1;
    for (int i = 0; i < n; i++) {
        map<pair<int,int>, int> slopes;
        int duplicates = 0, localBest = 0;
        for (int j = 0; j < n; j++) {
            if (i == j) continue;
            int dx = X[j] - X[i], dy = Y[j] - Y[i];
            if (dx == 0 && dy == 0) { duplicates++; continue; }
            int g = gcdSafe(dx, dy);
            dx /= g; dy /= g;
            if (dx < 0 || (dx == 0 && dy < 0)) { dx = -dx; dy = -dy; }
            slopes[{dx, dy}]++;
            localBest = max(localBest, slopes[{dx, dy}]);
        }
        best = max(best, localBest + duplicates + 1);
    }
    return best;
}
```
**Complexity:** O(n²).

### 1.3 String compression (run-length style)
**Problem:** `aabbbbeeeeffggg` → `a2b4e4f2g3`; `abbccccc` → `ab2c5` (count omitted when it's 1).

```cpp
string compress(string s) {
    string result;
    int n = s.size(), i = 0;
    while (i < n) {
        int j = i;
        while (j < n && s[j] == s[i]) j++;
        int count = j - i;
        result += s[i];
        if (count > 1) result += to_string(count);
        i = j;
    }
    return result;
}
```
**Complexity:** O(n).

### 1.6 Minimize the max−min difference (±k adjustment)
```cpp
int minimizeMaxDiff(vector<int> arr, int k) {
    sort(arr.begin(), arr.end());
    int n = arr.size();
    int ans = arr[n - 1] - arr[0];
    for (int i = 0; i < n - 1; i++) {
        int high = max(arr[i] + k, arr[n - 1] - k);
        int low = min(arr[0] + k, arr[i + 1] - k);
        ans = min(ans, high - low);
    }
    return ans;
}
```
**Complexity:** O(n log n).

### 1.7 Smallest window containing all characters of another string
**Problem:** S="abdd", X="bd" → smallest window = "bd".

```cpp
string smallestWindow(string s, string x) {
    unordered_map<char,int> need;
    for (char c : x) need[c]++;
    int missing = need.size();
    int left = 0, bestStart = 0, bestLen = INT_MAX;
    for (int right = 0; right < (int)s.size(); right++) {
        char ch = s[right];
        if (need.count(ch)) {
            need[ch]--;
            if (need[ch] == 0) missing--;
        }
        while (missing == 0) {
            if (right - left + 1 < bestLen) {
                bestLen = right - left + 1;
                bestStart = left;
            }
            char lch = s[left];
            if (need.count(lch)) {
                need[lch]++;
                if (need[lch] > 0) missing++;
            }
            left++;
        }
    }
    return bestLen == INT_MAX ? "" : s.substr(bestStart, bestLen);
}
```
**Complexity:** O(n).

### 1.8 Longest substring without repeating characters
```cpp
int longestUniqueSubstring(string s) {
    unordered_map<char,int> lastSeen;
    int start = 0, best = 0;
    for (int i = 0; i < (int)s.size(); i++) {
        char ch = s[i];
        if (lastSeen.count(ch) && lastSeen[ch] >= start) {
            start = lastSeen[ch] + 1;
        }
        lastSeen[ch] = i;
        best = max(best, i - start + 1);
    }
    return best;
}
```
**Complexity:** O(n).

### 1.12 Sort an array of only 0s, 1s, 2s (Dutch National Flag)
```cpp
void sort012(vector<int>& arr) {
    int low = 0, mid = 0, high = arr.size() - 1;
    while (mid <= high) {
        if (arr[mid] == 0) {
            swap(arr[low], arr[mid]);
            low++; mid++;
        } else if (arr[mid] == 1) {
            mid++;
        } else {
            swap(arr[mid], arr[high]);
            high--;
        }
    }
}
```

### 1.15 Maximal square of 1s in a binary matrix
```cpp
int maximalSquare(vector<vector<int>>& matrix) {
    if (matrix.empty() || matrix[0].empty()) return 0;
    int rows = matrix.size(), cols = matrix[0].size();
    vector<vector<int>> dp(rows + 1, vector<int>(cols + 1, 0));
    int bestSide = 0;
    for (int i = 1; i <= rows; i++) {
        for (int j = 1; j <= cols; j++) {
            if (matrix[i-1][j-1] == 1) {
                dp[i][j] = min({dp[i-1][j], dp[i][j-1], dp[i-1][j-1]}) + 1;
                bestSide = max(bestSide, dp[i][j]);
            }
        }
    }
    return bestSide * bestSide; // area of the largest square
}
```
**Complexity:** O(rows × cols).

### 1.16 Smallest divisor given a threshold
```cpp
int smallestDivisor(vector<int>& nums, int threshold) {
    int lo = 1, hi = *max_element(nums.begin(), nums.end());
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        long long total = 0;
        for (int x : nums) total += (x + mid - 1) / mid; // ceil division
        if (total <= threshold) hi = mid;
        else lo = mid + 1;
    }
    return lo;
}
```
**Complexity:** O(n log(max(nums))) — binary search on the answer.

### 2.1 Left view of a binary tree
```cpp
vector<int> leftView(TreeNode* root) {
    vector<int> result;
    if (!root) return result;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {
        int levelSize = q.size();
        for (int i = 0; i < levelSize; i++) {
            TreeNode* node = q.front(); q.pop();
            if (i == 0) result.push_back(node->val);
            if (node->left) q.push(node->left);
            if (node->right) q.push(node->right);
        }
    }
    return result;
}
```

### 2.3 Kth (or second) smallest element in a BST
```cpp
int kthSmallest(TreeNode* root, int k) {
    stack<TreeNode*> st;
    TreeNode* node = root;
    int count = 0;
    while (!st.empty() || node) {
        while (node) {
            st.push(node);
            node = node->left;
        }
        node = st.top(); st.pop();
        count++;
        if (count == k) return node->val;
        node = node->right;
    }
    return -1;
}
```

### 2.4 Build a reachability matrix from a tree (descendant matrix)
**Problem:** Given a tree, build a matrix M where M[i][j] = 1 if node j is a *descendant* of node i (a transitive ancestor–descendant matrix — confirmed by cross-checking the sample matrix in the source, consistent with the "Hint: Preorder" given alongside it).

```cpp
vector<int> collectDescendants(TreeNode* node) {
    vector<int> result;
    if (!node) return result;
    for (TreeNode* child : {node->left, node->right}) {
        if (child) {
            result.push_back(child->val);
            vector<int> sub = collectDescendants(child);
            result.insert(result.end(), sub.begin(), sub.end());
        }
    }
    return result;
}

void dfsFill(TreeNode* node, vector<vector<int>>& matrix) {
    if (!node) return;
    for (int d : collectDescendants(node)) matrix[node->val][d] = 1;
    dfsFill(node->left, matrix);
    dfsFill(node->right, matrix);
}

vector<vector<int>> descendantMatrix(TreeNode* root, int n) {
    vector<vector<int>> matrix(n + 1, vector<int>(n + 1, 0));
    dfsFill(root, matrix);
    return matrix;
}
```
**Complexity:** O(n²) worst case (skewed tree).

ListNode* reverseList(ListNode* head) {
    ListNode* prev = nullptr;
    while (head) {

### 3.2 Implement a stack using a linked list
```cpp
struct StackNode {
    int val;
    StackNode* next;
    StackNode(int x) : val(x), next(nullptr) {}
};

class Stack {
    StackNode* topNode;
    int sz;
public:
    Stack() : topNode(nullptr), sz(0) {}

    int getSize() { return sz; }
    bool isEmpty() { return sz == 0; }

    void push(int val) {
        StackNode* node = new StackNode(val);
        node->next = topNode;
        topNode = node;
        sz++;
    }

    void pop() {
        if (isEmpty()) return;
        StackNode* temp = topNode;
        topNode = topNode->next;
        delete temp;
        sz--;
    }

    int getTop() {
        return topNode ? topNode->val : -1;
    }
};
```
All operations O(1).

### 3.3 Minimum merges to make an array a palindrome
**Problem:** A=[1,2,3,4,5]; merging (summing) adjacent 1-indexed values 2 and 3 gives [1,5,4,5]. Find the minimum number of adjacent merges to make the array a palindrome.

```cpp
int minOpsToPalindrome(vector<int> arr) {
    int left = 0, right = arr.size() - 1;
    int ops = 0;
    long long leftSum = arr[left], rightSum = arr[right];
    while (left < right) {
        if (leftSum == rightSum) {
            left++; right--;
            if (left < right) {
                leftSum = arr[left];
                rightSum = arr[right];
            }
        } else if (leftSum < rightSum) {
            left++;
            leftSum += arr[left];
            ops++;
        } else {
            right--;
            rightSum += arr[right];
            ops++;
        }
    }
    return ops;
}
```
**Complexity:** O(n).

### 3.4 Postfix to prefix conversion
```cpp
bool isOperator(char c) {
    return c == '+' || c == '-' || c == '*' || c == '/' || c == '^';
}

string postfixToPrefix(string expr) {
    stack<string> st;
    for (char ch : expr) {
        if (isOperator(ch)) {
            string op2 = st.top(); st.pop();
            string op1 = st.top(); st.pop();
            st.push(string(1, ch) + op1 + op2);
        } else {
            st.push(string(1, ch));
        }
    }
    return st.top();
}
```
**Complexity:** O(n).

### 3.5 Implement a queue using two stacks
```cpp
class QueueUsingStacks {
    stack<int> inStack, outStack;
public:
    void enqueue(int val) {
        inStack.push(val);
    }

    int dequeue() {
        if (outStack.empty()) {
            while (!inStack.empty()) {
                outStack.push(inStack.top());
                inStack.pop();
            }
        }
        if (outStack.empty()) return -1; // empty queue
        int val = outStack.top();
        outStack.pop();
        return val;
    }
};
```
**Complexity:** Amortized O(1) per operation.

### 4.6 Bleak number check
**Problem:** n is "bleak" if there's no x such that `x + popcount(x) == n`.

```cpp
int isBleak(int n) {
    for (int x = max(1, n - 32); x < n; x++) {
        if (x + __builtin_popcount(x) == n) return false;

}
```

## 7. Theoretical / Conceptual Questions Previously Asked

These are the actual recurring questions reported from SAP Labs OAs and interview rounds (technical MCQs plus the questions interviewers ask right after the OA), organized by subject with direct answers.

### OOP

**Q: What is inheritance, and why do we need it?**
Lets a derived class acquire the properties/behavior of a base class, establishing an "is-a" relationship. It enables code reuse and lets you extend or override behavior instead of rewriting it.

**Q: What are the types of inheritance?**
Single (one base → one derived), Multilevel (A→B→C), Hierarchical (one base → many derived), Multiple (one derived ← many bases — allowed in C++, not for classes in Java), and Hybrid (a combination of these).

**Q: Why doesn't Java support multiple inheritance of classes?**
To avoid the diamond problem — ambiguity when two parent classes define the same method and the compiler can't tell which one the child should use. Java sidesteps this by allowing a class to implement multiple interfaces instead.

**Q: What is polymorphism? How many types?**
The same interface/call behaving differently depending on context. Compile-time (method overloading — same name, different parameters) and run-time (method overriding — subclass provides its own implementation, resolved via dynamic dispatch).

**Q: Difference between abstraction and encapsulation?**
Abstraction hides implementation complexity, exposing only relevant behavior (achieved via abstract classes/interfaces). Encapsulation hides internal data by bundling it with methods and restricting access (achieved via access modifiers and getters/setters).

**Q: Abstract class vs interface — when do you use an abstract class?**
An abstract class can mix implemented and abstract methods plus hold state — use it when related classes share common code. An interface is a pure contract (default methods aside) — use it when unrelated classes just need to guarantee a capability.

**Q: What is multithreading?**
Running multiple threads concurrently within one process, sharing memory, to improve responsiveness and use multiple CPU cores.

**Q: Deep copy vs shallow copy?**
Shallow copy duplicates the top-level object but keeps references to nested objects (shared between copies). Deep copy recursively duplicates nested objects too, so both copies are fully independent.

**Q: Copy constructor vs the assignment operator (`operator=`)?**
Copy constructor initializes a *new* object as a copy of an existing one. `operator=` copies values into an *already-existing* object. Both must handle deep-copy semantics if the class owns dynamically allocated resources.

**Q: What is the Singleton pattern? (asked to implement it)**
Ensures a class has exactly one instance with a global access point — a private constructor, a static instance, and a static `getInstance()` (with thread-safety handled for multi-threaded use).

**Q: What is the Factory pattern?**
A creational pattern where a factory method creates and returns objects without exposing the exact instantiation logic to the caller, decoupling creation from use.

**Q: What is Dependency Injection, and which pattern supports it?**
A technique where an object's dependencies are supplied externally (constructor/setter/interface injection) instead of being created internally — an application of Inversion of Control, often paired with the Factory pattern to build the injected objects.

**Q: What is the Strategy pattern?**
Encapsulates a family of interchangeable algorithms behind a common interface, letting the algorithm be swapped at runtime without changing the client code.

**Q: Difference between an array and a linked list?**
Arrays: O(1) random access, O(n) insert/delete (shifting), contiguous/fixed-ish size. Linked lists: O(1) insert/delete at a known node, O(n) access (no indexing), extra pointer overhead.

### DBMS

**Q: What does the "I" in ACID stand for?**
Isolation — concurrent transactions behave as if executed sequentially; one transaction's intermediate state isn't visible to another.

**Q: What is write-ahead logging?**
Changes are recorded in a durable log *before* being applied to the actual data, so a crash can be recovered from by replaying/undoing the log — underpins atomicity and durability.

**Q: What are deadlock avoidance schemes in DBMS?**
Wait-die and wound-wait (timestamp-based rules deciding whether a transaction waits or is rolled back, based on which is older), timeout-based detection, and consistent resource-ordering.

**Q: What is a clustering index?**
An index where the table's physical row order matches the index key order — at most one per table — versus non-clustering (secondary) indexes, which point to rows without reordering the table, and a table can have several.

**Q: What are the normal forms (1NF/2NF/3NF)?**
1NF: atomic column values, no repeating groups. 2NF: 1NF + no partial dependency of non-key attributes on part of a composite key. 3NF: 2NF + no transitive dependency between non-key attributes.

**Q: Relational vs non-relational (NoSQL) databases?**
Relational: fixed schema, foreign-key relationships, strong ACID guarantees — good for structured transactional data. NoSQL (document/key-value/column/graph): flexible schema, scales horizontally more easily, often favors availability over strict consistency (eventual consistency).

**Q: MySQL vs MongoDB — which scales better and how?**
MySQL traditionally scales vertically and needs more engineering effort to shard horizontally due to joins/consistency constraints. MongoDB is built for horizontal scaling (sharding is first-class) at the cost of weaker cross-document transaction guarantees.

**Q: Why have column-oriented databases evolved?**
Analytical (OLAP) queries scan a few columns across huge row counts; storing data column-by-column lets those scans read only the needed columns and compress similar values well, making aggregations much faster than row-oriented storage.

**Q: Explain joins/normalization/aggregate functions (asked as a live SQL exercise on an employee table).**
See the SQL examples in Section 5 — a JOIN combines rows from related tables on a key, normalization removes redundancy per the forms above, and aggregate functions (`COUNT`, `SUM`, `AVG`, etc.) summarize grouped rows, usually paired with `GROUP BY`/`HAVING`.

### Operating Systems

**Q: Difference between a process and a thread?**
A process is an independent program in execution with its own memory space. A thread is a lightweight unit of execution *within* a process, sharing memory/resources with other threads of that process — cheaper to create/switch, less isolated.

**Q: Multitasking vs multiprocessing — and limitations of multiprocessing?**
Multitasking time-slices a single CPU across tasks (illusion of parallelism via context switching). Multiprocessing uses multiple CPUs/cores for true parallelism. Limitations: IPC/synchronization overhead, higher memory use (separate address spaces), and diminishing returns if the workload doesn't parallelize well.

**Q: Necessary conditions for deadlock?**
Mutual exclusion, hold-and-wait, no preemption, and circular wait — all four must hold simultaneously.

### Java-specific

**Q: Difference between JVM, JDK, and JRE?**
JVM executes bytecode, giving platform independence. JRE = JVM + core libraries needed to *run* Java apps. JDK = JRE + development tools (compiler, debugger, etc.) needed to *build* Java apps.

**Q: How is a HashMap implemented internally?**
An array of buckets; a key's (hash-spread) hashCode determines its bucket. Collisions within a bucket are handled with a linked list (or a balanced tree for large buckets, since Java 8), and `equals()` resolves exact matches within the bucket.

**Q: HashMap vs HashSet?**
HashMap stores key–value pairs. HashSet stores unique elements only — it's internally backed by a HashMap where the elements are the keys and a constant dummy value is used.

### Networking / Security

**Q: Explain public key cryptography.**
An asymmetric scheme using a linked key pair: a public key (shared openly, used to encrypt or verify signatures) and a private key (kept secret, used to decrypt or sign). Anyone can encrypt with the public key, but only the private-key holder can decrypt.

**Q: What is a digital certificate?**
A certificate, issued by a trusted Certificate Authority, that binds a public key to an identity (person/server/org), so others can verify the key genuinely belongs to that entity — the basis of trust in HTTPS/TLS.

### Version control

**Q: Difference between `git pull` and `git fetch`?**
`git fetch` downloads new commits/branches from the remote without touching your working branch — it just updates your local view of the remote. `git pull` does a fetch *and* automatically merges (or rebases) those changes into your current branch.

**Q: Design a database schema for an Instagram-like app, including comment threads.**
Tables for Users, Posts, and Comments; Comments has a self-referencing `parent_comment_id` to support threaded replies, plus `post_id` and `user_id` foreign keys. Index `post_id` and `parent_comment_id` for fast thread retrieval.

A User class, a Product class, a Cart holding a list of CartItems (each referencing a Product + quantity), and an Order created from a Cart at checkout — modeled with one-to-many/many-to-many relationships as needed.

**Q: Best practices for REST API design?**
Use nouns for resources and HTTP verbs for actions, keep endpoints stateless, use correct status codes, version the API, support pagination/filtering on collections, and keep response shapes consistent.

### Puzzles (asked in interview rounds, not the OA itself)

**Q: Find the odd (heavier/lighter) coin among a group using a balance scale in minimum weighings.**
Split the coins into three roughly equal groups, weigh two against each other. The unbalanced side (or, if balanced, the untested group) contains the odd coin — recurse on that smaller group. This ternary elimination needs about log₃(n) weighings for n coins.

**Q: Measure a specific time duration using items that burn/deplete unevenly (e.g., candles/ropes).**
Light a candle/rope from both ends (or two at once) to halve its burn time versus lighting from one end, and chain these halvings/combinations to hit durations you can't time directly.

**Q: Maximum pieces from a fixed number of straight cuts through a cake (top-down view).**
For n straight cuts, the maximum pieces = 1 + n + C(n, 2) — each new cut can cross every previous cut, and each new crossing adds one extra piece.
